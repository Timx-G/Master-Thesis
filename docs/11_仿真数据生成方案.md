# 11 仿真数据生成方案

本文档说明如何生成V2X-MEC卸载实验所需的仿真数据，**不需要真实数据或开源数据集**。

---

## 1. 为什么不需要真实数据？

### 1.1 学术界的标准做法

**V2X-MEC领域的论文普遍使用仿真数据**：

| 顶会/顶刊 | 论文示例 | 数据来源 |
|---------|---------|---------|
| IEEE TVT | 95%以上论文 | SUMO仿真 + 参数化模型 |
| IEEE TMC | 90%以上论文 | 自研仿真器 |
| MobiCom | 80%以上论文 | NS-3 + 自定义脚本 |

**真实数据的问题**：
- ❌ **不存在**：V2X计算卸载的真实数据集几乎没有公开
- ❌ **难获取**：涉及运营商、车企的敏感数据
- ❌ **不可控**：无法调整参数做对比实验

### 1.2 审稿人的期待

审稿人**不会要求真实数据**，只要求：
1. ✅ **参数来源可追溯**：引用3GPP/ETSI标准
2. ✅ **场景设置合理**：城市道路、车辆密度符合实际
3. ✅ **可复现**：提供仿真代码和参数

### 2.2 数据生成模块

#### **模块1：车辆轨迹生成**

```python
"""
trajectory_generator.py - 车辆轨迹生成器
"""

import numpy as np
from dataclasses import dataclass
from typing import List

@dataclass
class Vehicle:
    id: int
    position: np.ndarray  # [x, y]
    speed: float          # m/s
    heading: float        # 航向角（弧度）
    lane: int

class TrajectoryGenerator:
    """
    基于简化城市道路模型的轨迹生成器
    """

    def __init__(self,
                 road_length: float = 2000,  # 道路长度（m）
                 num_lanes: int = 3,
                 speed_limit_urban: float = 50/3.6,  # 城市：50 km/h
                 speed_limit_highway: float = 120/3.6):  # 高速：120 km/h
        self.road_length = road_length
        self.num_lanes = num_lanes
        self.speed_limit_urban = speed_limit_urban
        self.speed_limit_highway = speed_limit_highway

    def generate_vehicles(self, num_vehicles: int, scenario: str = 'urban') -> List[Vehicle]:
        """
        生成初始车辆分布

        参数基于：
        - 3GPP TR 37.885（V2X场景）
        - ETSI TR 103 562（CAM/CPM场景）
        """
        vehicles = []

        if scenario == 'urban':
            # 城市场景：速度30-50 km/h
            speed_mean = 40 / 3.6  # m/s
            speed_std = 10 / 3.6
        elif scenario == 'highway':
            # 高速场景：速度80-120 km/h
            speed_mean = 100 / 3.6
            speed_std = 20 / 3.6
        else:
            raise ValueError(f"Unknown scenario: {scenario}")

        for i in range(num_vehicles):
            # 随机初始位置（沿道路均匀分布）
            x = np.random.uniform(0, self.road_length)
            lane = np.random.randint(0, self.num_lanes)
            y = lane * 3.5  # 车道宽度3.5m（标准）

            # 速度（正态分布）
            speed = np.random.normal(speed_mean, speed_std)
            speed = np.clip(speed, 5, self.speed_limit_urban if scenario == 'urban' else self.speed_limit_highway)

            # 航向角（沿道路方向）
            heading = 0.0  # 简化为直线道路

            vehicles.append(Vehicle(
                id=i,
                position=np.array([x, y]),
                speed=speed,
                heading=heading,
                lane=lane
            ))

        return vehicles

    def update_positions(self, vehicles: List[Vehicle], dt: float = 0.1):
        """
        更新车辆位置（匀速直线运动 + 随机扰动）

        Args:
            vehicles: 车辆列表
            dt: 时间步长（秒）
        """
        for v in vehicles:
            # 位置更新
            dx = v.speed * dt
            v.position[0] += dx

            # 循环道路（到达终点后重置到起点）
            if v.position[0] > self.road_length:
                v.position[0] -= self.road_length

            # 速度微小波动（模拟真实驾驶）
            v.speed += np.random.normal(0, 0.5) * dt
            v.speed = np.clip(v.speed, 5, self.speed_limit_urban)
```

#### **模块2：CPM任务生成器**

```python
"""
cpm_generator.py - CPM任务生成器
基于ETSI TS 103 324标准
"""

import numpy as np
from typing import List, Optional

class CPM_TaskGenerator:
    """
    事件触发CPM任务生成器
    """

    def __init__(self,
                 T_check: float = 0.1,           # 检查周期100ms
                 pos_threshold: float = 4.0,      # 位置阈值4m
                 speed_threshold: float = 0.5,    # 速度阈值0.5m/s
                 heading_threshold: float = 4.0,  # 航向阈值4度
                 T_max: float = 1.0):             # 最大间隔1s

        self.T_check = T_check
        self.pos_threshold = pos_threshold
        self.speed_threshold = speed_threshold
        self.heading_threshold = np.deg2rad(heading_threshold)
        self.T_max = T_max

        # 记录上次触发状态
        self.last_trigger_state = {}

    def check_trigger(self, vehicle: Vehicle, current_time: float) -> bool:
        """
        检查是否触发CPM生成

        Returns:
            True: 触发，False: 不触发
        """
        vid = vehicle.id

        # 首次触发
        if vid not in self.last_trigger_state:
            self.last_trigger_state[vid] = {
                'position': vehicle.position.copy(),
                'speed': vehicle.speed,
                'heading': vehicle.heading,
                'time': current_time
            }
            return True

        last_state = self.last_trigger_state[vid]

        # 条件1：位置变化 >= 4m
        pos_change = np.linalg.norm(vehicle.position - last_state['position'])
        if pos_change >= self.pos_threshold:
            self._update_state(vid, vehicle, current_time)
            return True

        # 条件2：速度变化 >= 0.5 m/s
        speed_change = abs(vehicle.speed - last_state['speed'])
        if speed_change >= self.speed_threshold:
            self._update_state(vid, vehicle, current_time)
            return True

        # 条件3：航向变化 >= 4度
        heading_change = abs(vehicle.heading - last_state['heading'])
        if heading_change >= self.heading_threshold:
            self._update_state(vid, vehicle, current_time)
            return True

        # 条件5：时间间隔 >= T_max
        time_elapsed = current_time - last_state['time']
        if time_elapsed >= self.T_max:
            self._update_state(vid, vehicle, current_time)
            return True

        return False

    def _update_state(self, vid: int, vehicle: Vehicle, current_time: float):
        """更新上次触发状态"""
        self.last_trigger_state[vid] = {
            'position': vehicle.position.copy(),
            'speed': vehicle.speed,
            'heading': vehicle.heading,
            'time': current_time
        }

    def generate_task(self, vehicle: Vehicle) -> dict:
        """
        生成CPM任务

        任务大小参考：
        - ETSI TS 103 324: CPM消息大小300-800 bytes
        - 计算负载：目标检测、跟踪、融合（经验值）
        """
        # 数据大小（bits）
        # 基础CPM: 300-800 bytes
        # 传感器数据: 根据目标数量变化
        num_objects = np.random.randint(5, 20)  # 检测到的目标数量
        data_size = (500 + num_objects * 50) * 8  # bytes转bits

        # 计算负载（CPU cycles）
        # 参考文献：目标检测约1e8-1e9 cycles
        compute_load = np.random.uniform(5e8, 1e9)

        # 截止时间（根据AoI要求）
        deadline = 0.2  # 200ms（协作感知的实时性要求）

        # 结果数据大小（小）
        result_size = 100 * 8  # 100 bytes

        return {
            'vehicle_id': vehicle.id,
            'data_size': data_size,
            'compute_load': compute_load,
            'deadline': deadline,
            'result_size': result_size
        }
```

#### **模块3：信道模型**

```python
"""
channel_model.py - V2I信道模型
基于3GPP TR 37.885
"""

import numpy as np

class V2I_ChannelModel:
    """
    3GPP Urban场景下的V2I信道模型
    """

    def __init__(self,
                 carrier_freq: float = 5.9e9,    # 5.9 GHz (C-V2X)
                 bandwidth: float = 20e6,        # 20 MHz
                 tx_power_dbm: float = 23,       # 23 dBm (200 mW)
                 noise_power_dbm: float = -114): # 热噪声

        self.carrier_freq = carrier_freq
        self.bandwidth = bandwidth
        self.tx_power_dbm = tx_power_dbm
        self.noise_power_dbm = noise_power_dbm

    def path_loss(self, distance: float) -> float:
        """
        路径损耗（dB）

        模型：3GPP TR 37.885 Urban NLoS
        PL = 36.7 * log10(d) + 22.7 + 26 * log10(fc)
        """
        if distance < 1:
            distance = 1  # 避免log(0)

        fc_ghz = self.carrier_freq / 1e9
        pl = 36.7 * np.log10(distance) + 22.7 + 26 * np.log10(fc_ghz)

        # 添加阴影衰落（对数正态）
        shadowing = np.random.normal(0, 4)  # 标准差4dB

        return pl + shadowing

    def compute_rate(self, vehicle_pos: np.ndarray, rsu_pos: np.ndarray) -> float:
        """
        计算传输速率（bps）

        使用Shannon公式：R = B * log2(1 + SNR)
        """
        distance = np.linalg.norm(vehicle_pos - rsu_pos)

        # 路径损耗
        pl_db = self.path_loss(distance)

        # 接收功率（dBm）
        rx_power_dbm = self.tx_power_dbm - pl_db

        # SNR（dB）
        snr_db = rx_power_dbm - self.noise_power_dbm
        snr_linear = 10 ** (snr_db / 10)

        # Shannon速率
        rate = self.bandwidth * np.log2(1 + snr_linear)

        return rate  # bps
```

## 3. 进阶方案：SUMO + Python（更真实）

### 3.1 为什么用SUMO？

**SUMO (Simulation of Urban MObility)** 是交通仿真的黄金标准：

✅ **审稿人认可度高**：被IEEE TVT/TMC/TITS广泛接受
✅ **真实交通流**：包含加速、减速、换道、跟车模型
✅ **预定义场景**：提供城市、高速、十字路口等标准场景

```bash

    --trip-attributes="departSpeed=\"max\"" -e 100
```
