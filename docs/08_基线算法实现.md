# 08 基线算法实现

## 概述

为了验证所提算法的有效性，需要实现多个基线算法进行对比实验。本文档提供6种基线算法的完整实现。

---

## 1. 基线算法列表

| 算法名称 | 缩写 | 核心思想 | 用于验证 |
|---------|------|---------|---------|
| Local Only | LO | 所有任务本地计算 | 卸载的必要性 |
| Full Offload | FO | 所有任务卸载到边缘 | 选择性卸载的优势 |
| Greedy Offload | GO | 贪心选择最小时延方案 | 长期优化的优势 |
| Random Offload | RO | 随机选择卸载决策 | 智能决策的必要性 |
| Poisson Model | PM | 假设泊松到达的传统方法 | 周期性建模的优势 |
| No Phase Align | NPA | 无相位对齐的周期性卸载 | 相位对齐的必要性 |
| **DQN** | DQN | 深度Q网络强化学习 | **与主流DRL方法对比** |
| **PPO** | PPO | 近端策略优化 | **与主流DRL方法对比** |

**重要说明**：DQN和PPO是审稿人期待的关键基线，用于证明Lyapunov方法在V2X场景下的优势。详细实现见 [08.1_DRL基线算法实现.md](./08.1_DRL基线算法实现.md)

## 2. 基线算法基类

```python
"""
baseline_base.py - 基线算法基类
"""

from abc import ABC, abstractmethod
from dataclasses import dataclass
from typing import List, Dict, Tuple, Optional
import numpy as np

@dataclass
class Task:
    """任务数据结构"""
    task_id: str
    vehicle_id: str
    data_size: float          # bits
    computation_load: float    # CPU cycles
    deadline: float           # seconds
    generation_time: float    # 生成时间
    period: float             # 周期
    priority: int = 1         # 优先级

@dataclass
class OffloadingDecision:
    """卸载决策"""
    task_id: str
    offload: bool             # True=卸载, False=本地
    target_rsu: Optional[str] = None  # 目标RSU
    allocated_bandwidth: float = 0.0
    allocated_computation: float = 0.0

@dataclass
class ExecutionResult:
    """执行结果"""
    task_id: str
    completion_time: float
    energy_consumption: float
    aoi: float
    success: bool             # 是否在deadline内完成

class BaselineAlgorithm(ABC):
    """基线算法抽象基类"""

    def __init__(self, config: Dict):
        self.config = config
        self.name = "BaselineAlgorithm"

    @abstractmethod
    def make_decision(self,
                     tasks: List[Task],
                     system_state: Dict) -> List[OffloadingDecision]:
        """
        做出卸载决策

        Args:
            tasks: 待处理任务列表
            system_state: 系统状态（信道、负载等）

        Returns:
            卸载决策列表
        """
        pass

    def calculate_local_time(self, task: Task, f_local: float) -> float:
        """计算本地执行时间"""
        return task.computation_load / f_local

    def calculate_offload_time(self, task: Task,
                               bandwidth: float,
                               channel_gain: float,
                               transmit_power: float,
                               noise_power: float,
                               f_edge: float) -> Tuple[float, float]:
        """
        计算卸载执行时间

        Returns:
            (传输时间, 计算时间)
        """
        # 传输速率 (Shannon公式)
        snr = transmit_power * channel_gain / noise_power
        rate = bandwidth * np.log2(1 + snr)

        # 传输时间
        trans_time = task.data_size / rate if rate > 0 else float('inf')

        # 计算时间
        comp_time = task.computation_load / f_edge

        return trans_time, comp_time

    def calculate_local_energy(self, task: Task,
                               f_local: float,
                               kappa: float = 1e-27) -> float:
        """计算本地执行能耗"""
        return kappa * task.computation_load * (f_local ** 2)

    def calculate_offload_energy(self, task: Task,
                                 transmit_power: float,
                                 trans_time: float) -> float:
        """计算卸载能耗（仅传输）"""
        return transmit_power * trans_time
```

## 3. Local Only (LO) - 纯本地计算

```python
"""
baseline_local_only.py - 所有任务本地执行
"""

class LocalOnlyAlgorithm(BaselineAlgorithm):
    """
    Local Only: 所有任务在本地执行

    用途：验证计算卸载的必要性
    预期：高时延、高能耗、低任务完成率
    """

    def __init__(self, config: Dict):
        super().__init__(config)
        self.name = "Local Only"

    def make_decision(self,
                     tasks: List[Task],
                     system_state: Dict) -> List[OffloadingDecision]:
        """所有任务选择本地执行"""
        decisions = []

        for task in tasks:
            decision = OffloadingDecision(
                task_id=task.task_id,
                offload=False,  # 不卸载
                target_rsu=None,
                allocated_bandwidth=0.0,
                allocated_computation=0.0
            )
            decisions.append(decision)

        return decisions

    def execute(self, task: Task, config: Dict) -> ExecutionResult:
        """执行任务"""
        f_local = config.get('f_local', 1e9)  # 1 GHz
        kappa = config.get('kappa', 1e-27)

        # 计算时间和能耗
        exec_time = self.calculate_local_time(task, f_local)
        energy = self.calculate_local_energy(task, f_local, kappa)

        # 判断是否成功
        success = exec_time <= task.deadline

        # 计算AoI
        aoi = exec_time if success else task.deadline + exec_time

        return ExecutionResult(
            task_id=task.task_id,
            completion_time=exec_time,
            energy_consumption=energy,
            aoi=aoi,
            success=success
        )
```

## 4. Full Offload (FO) - 完全卸载

```python
"""
baseline_full_offload.py - 所有任务卸载到边缘
"""

class FullOffloadAlgorithm(BaselineAlgorithm):
    """
    Full Offload: 所有任务卸载到边缘服务器

    用途：验证选择性卸载的优势
    预期：高能耗（传输）、可能的排队时延、带宽竞争
    """

    def __init__(self, config: Dict):
        super().__init__(config)
        self.name = "Full Offload"

    def make_decision(self,
                     tasks: List[Task],
                     system_state: Dict) -> List[OffloadingDecision]:
        """所有任务选择卸载"""
        decisions = []

        # 获取可用RSU列表
        available_rsus = system_state.get('available_rsus', ['RSU_0'])
        total_bandwidth = system_state.get('total_bandwidth', 20e6)  # 20 MHz
        total_computation = system_state.get('total_computation', 10e9)  # 10 GHz

        # 公平分配带宽和计算资源
        n_tasks = len(tasks)
        if n_tasks > 0:
            bandwidth_per_task = total_bandwidth / n_tasks
            computation_per_task = total_computation / n_tasks
        else:
            bandwidth_per_task = total_bandwidth
            computation_per_task = total_computation

        for task in tasks:
            # 选择最近的RSU
            vehicle_pos = system_state.get('vehicle_positions', {}).get(task.vehicle_id, (0, 0))
            rsu_positions = system_state.get('rsu_positions', {'RSU_0': (0, 0)})

            # 找最近RSU
            min_dist = float('inf')
            nearest_rsu = available_rsus[0]
            for rsu_id, rsu_pos in rsu_positions.items():
                dist = np.sqrt((vehicle_pos[0] - rsu_pos[0])**2 +
                              (vehicle_pos[1] - rsu_pos[1])**2)
                if dist < min_dist:
                    min_dist = dist
                    nearest_rsu = rsu_id

            decision = OffloadingDecision(
                task_id=task.task_id,
                offload=True,  # 全部卸载
                target_rsu=nearest_rsu,
                allocated_bandwidth=bandwidth_per_task,
                allocated_computation=computation_per_task
            )
            decisions.append(decision)

        return decisions

    def execute(self, task: Task, decision: OffloadingDecision,
                system_state: Dict, config: Dict) -> ExecutionResult:
        """执行任务"""
        # 获取参数
        channel_gain = system_state.get('channel_gains', {}).get(
            (task.vehicle_id, decision.target_rsu), 1e-4)
        transmit_power = config.get('transmit_power', 0.5)  # 500 mW
        noise_power = config.get('noise_power', 1e-13)

        # 计算时延
        trans_time, comp_time = self.calculate_offload_time(
            task,
            decision.allocated_bandwidth,
            channel_gain,
            transmit_power,
            noise_power,
            decision.allocated_computation
        )

        total_time = trans_time + comp_time

        # 计算能耗
        energy = self.calculate_offload_energy(task, transmit_power, trans_time)

        # 判断成功
        success = total_time <= task.deadline
        aoi = total_time if success else task.deadline + total_time

        return ExecutionResult(
            task_id=task.task_id,
            completion_time=total_time,
            energy_consumption=energy,
            aoi=aoi,
            success=success
        )
```

## 5. Greedy Offload (GO) - 贪心卸载

```python
"""
baseline_greedy.py - 贪心选择最小时延方案
"""

class GreedyOffloadAlgorithm(BaselineAlgorithm):
    """
    Greedy Offload: 每个任务贪心选择当前时延最小的方案

    用途：验证长期优化（Lyapunov）的优势
    预期：短期性能好，但长期可能导致负载不均、能耗波动
    """

    def __init__(self, config: Dict):
        super().__init__(config)
        self.name = "Greedy Offload"

    def make_decision(self,
                     tasks: List[Task],
                     system_state: Dict) -> List[OffloadingDecision]:
        """贪心选择最小时延方案"""
        decisions = []

        # 复制资源状态用于动态更新
        remaining_bandwidth = system_state.get('total_bandwidth', 20e6)
        remaining_computation = system_state.get('total_computation', 10e9)

        f_local = self.config.get('f_local', 1e9)
        transmit_power = self.config.get('transmit_power', 0.5)
        noise_power = self.config.get('noise_power', 1e-13)

        for task in tasks:
            # 计算本地执行时间
            local_time = self.calculate_local_time(task, f_local)

            # 计算卸载执行时间（假设能获得一定比例的资源）
            n_remaining = len(tasks) - len(decisions)
            if n_remaining > 0:
                available_bw = remaining_bandwidth / n_remaining
                available_comp = remaining_computation / n_remaining
            else:
                available_bw = remaining_bandwidth
                available_comp = remaining_computation

            # 获取信道增益
            vehicle_id = task.vehicle_id
            rsu_id = system_state.get('available_rsus', ['RSU_0'])[0]
            channel_gain = system_state.get('channel_gains', {}).get(
                (vehicle_id, rsu_id), 1e-4)

            trans_time, comp_time = self.calculate_offload_time(
                task, available_bw, channel_gain,
                transmit_power, noise_power, available_comp
            )
            offload_time = trans_time + comp_time

            # 贪心选择
            if offload_time < local_time and remaining_bandwidth > 0:
                # 选择卸载
                decision = OffloadingDecision(
                    task_id=task.task_id,
                    offload=True,
                    target_rsu=rsu_id,
                    allocated_bandwidth=available_bw,
                    allocated_computation=available_comp
                )
                # 更新剩余资源
                remaining_bandwidth -= available_bw
                remaining_computation -= available_comp
            else:
                # 选择本地
                decision = OffloadingDecision(
                    task_id=task.task_id,
                    offload=False,
                    target_rsu=None,
                    allocated_bandwidth=0.0,
                    allocated_computation=0.0
                )

            decisions.append(decision)

        return decisions
```

## 6. Random Offload (RO) - 随机卸载

```python
"""
baseline_random.py - 随机卸载决策
"""

class RandomOffloadAlgorithm(BaselineAlgorithm):
    """
    Random Offload: 随机选择卸载决策

    用途：作为下界基准，验证智能决策的必要性
    预期：性能最差，波动大
    """

    def __init__(self, config: Dict, seed: int = 42):
        super().__init__(config)
        self.name = "Random Offload"
        self.rng = np.random.RandomState(seed)

    def make_decision(self,
                     tasks: List[Task],
                     system_state: Dict) -> List[OffloadingDecision]:
        """随机决策"""
        decisions = []

        total_bandwidth = system_state.get('total_bandwidth', 20e6)
        total_computation = system_state.get('total_computation', 10e9)
        available_rsus = system_state.get('available_rsus', ['RSU_0'])

        offload_count = 0

        for task in tasks:
            # 50%概率卸载
            offload = self.rng.random() < 0.5

            if offload:
                offload_count += 1
                # 随机选择RSU
                target_rsu = self.rng.choice(available_rsus)

                decision = OffloadingDecision(
                    task_id=task.task_id,
                    offload=True,
                    target_rsu=target_rsu,
                    allocated_bandwidth=0.0,  # 稍后分配
                    allocated_computation=0.0
                )
            else:
                decision = OffloadingDecision(
                    task_id=task.task_id,
                    offload=False,
                    target_rsu=None,
                    allocated_bandwidth=0.0,
                    allocated_computation=0.0
                )

            decisions.append(decision)

        # 公平分配资源给卸载任务
        if offload_count > 0:
            bw_per_task = total_bandwidth / offload_count
            comp_per_task = total_computation / offload_count

            for decision in decisions:
                if decision.offload:
                    decision.allocated_bandwidth = bw_per_task
                    decision.allocated_computation = comp_per_task

        return decisions
```

## 7. Poisson Model (PM) - 泊松到达模型

```python
"""
baseline_poisson.py - 基于泊松到达假设的传统方法
"""

class PoissonModelAlgorithm(BaselineAlgorithm):
    """
    Poisson Model: 假设任务按泊松过程到达

    用途：验证周期性建模相比传统泊松假设的优势
    预期：无法准确预测任务到达，资源分配效率低
    """

    def __init__(self, config: Dict):
        super().__init__(config)
        self.name = "Poisson Model"
        self.lambda_estimated = config.get('poisson_rate', 10.0)  # 估计的到达率
        self.history = []  # 历史记录用于更新估计

    def update_arrival_rate(self, observed_arrivals: int, time_window: float):
        """更新泊松到达率估计"""
        if time_window > 0:
            observed_rate = observed_arrivals / time_window
            # 指数平滑更新
            alpha = 0.3
            self.lambda_estimated = alpha * observed_rate + (1 - alpha) * self.lambda_estimated

    def predict_arrivals(self, time_window: float) -> float:
        """预测未来到达数量"""
        return self.lambda_estimated * time_window

    def make_decision(self,
                     tasks: List[Task],
                     system_state: Dict) -> List[OffloadingDecision]:
        """基于泊松预测的决策"""
        decisions = []

        # 预测下一时隙的任务数量
        slot_duration = self.config.get('slot_duration', 0.1)
        predicted_tasks = self.predict_arrivals(slot_duration)

        # 基于预测分配资源
        total_bandwidth = system_state.get('total_bandwidth', 20e6)
        total_computation = system_state.get('total_computation', 10e9)

        # 预留资源给预测的任务
        expected_total = max(len(tasks), predicted_tasks)

        f_local = self.config.get('f_local', 1e9)
        transmit_power = self.config.get('transmit_power', 0.5)
        noise_power = self.config.get('noise_power', 1e-13)

        for task in tasks:
            # 计算本地和卸载时延
            local_time = self.calculate_local_time(task, f_local)

            # 基于泊松预测分配资源
            if expected_total > 0:
                available_bw = total_bandwidth / expected_total
                available_comp = total_computation / expected_total
            else:
                available_bw = total_bandwidth
                available_comp = total_computation

            # 获取信道信息
            rsu_id = system_state.get('available_rsus', ['RSU_0'])[0]
            channel_gain = system_state.get('channel_gains', {}).get(
                (task.vehicle_id, rsu_id), 1e-4)

            trans_time, comp_time = self.calculate_offload_time(
                task, available_bw, channel_gain,
                transmit_power, noise_power, available_comp
            )
            offload_time = trans_time + comp_time

            # 简单阈值决策
            if offload_time < local_time * 0.8:  # 卸载时延需要比本地低20%
                decision = OffloadingDecision(
                    task_id=task.task_id,
                    offload=True,
                    target_rsu=rsu_id,
                    allocated_bandwidth=available_bw,
                    allocated_computation=available_comp
                )
            else:
                decision = OffloadingDecision(
                    task_id=task.task_id,
                    offload=False,
                    target_rsu=None,
                    allocated_bandwidth=0.0,
                    allocated_computation=0.0
                )

            decisions.append(decision)

        # 更新到达率估计
        self.update_arrival_rate(len(tasks), slot_duration)

        return decisions
```

## 8. No Phase Align (NPA) - 无相位对齐

```python
"""
baseline_no_phase_align.py - 无相位对齐的周期性卸载
"""

class NoPhaseAlignAlgorithm(BaselineAlgorithm):
    """
    No Phase Align: 保留周期性建模但不进行相位对齐

    用途：验证相位对齐调度的必要性
    预期：周期开始时刻任务同步到达，造成严重拥塞
    """

    def __init__(self, config: Dict):
        super().__init__(config)
        self.name = "No Phase Align"
        self.task_periods = {}  # 记录每个车辆的任务周期

    def register_task_period(self, vehicle_id: str, period: float, phase: float = 0.0):
        """注册任务周期（不调整相位）"""
        self.task_periods[vehicle_id] = {
            'period': period,
            'phase': phase,  # 保持原始相位，不对齐
            'last_generation': 0.0
        }

    def predict_next_generation(self, vehicle_id: str, current_time: float) -> float:
        """预测下一次任务生成时间"""
        if vehicle_id not in self.task_periods:
            return current_time

        info = self.task_periods[vehicle_id]
        period = info['period']
        phase = info['phase']

        # 计算下一个周期开始时间
        cycles_passed = int((current_time - phase) / period)
        next_time = phase + (cycles_passed + 1) * period

        return next_time

    def make_decision(self,
                     tasks: List[Task],
                     system_state: Dict) -> List[OffloadingDecision]:
        """不进行相位对齐的决策"""
        decisions = []

        total_bandwidth = system_state.get('total_bandwidth', 20e6)
        total_computation = system_state.get('total_computation', 10e9)

        f_local = self.config.get('f_local', 1e9)
        transmit_power = self.config.get('transmit_power', 0.5)
        noise_power = self.config.get('noise_power', 1e-13)

        # 识别同步到达的任务（周期性同步问题）
        current_time = system_state.get('current_time', 0.0)

        # 统计当前时隙的任务数量（体现同步拥塞）
        concurrent_tasks = len(tasks)

        # 资源分配（由于同步，资源竞争激烈）
        if concurrent_tasks > 0:
            bw_per_task = total_bandwidth / concurrent_tasks
            comp_per_task = total_computation / concurrent_tasks
        else:
            bw_per_task = total_bandwidth
            comp_per_task = total_computation

        for task in tasks:
            # 注册周期
            self.register_task_period(task.vehicle_id, task.period)

            # 计算时延
            local_time = self.calculate_local_time(task, f_local)

            rsu_id = system_state.get('available_rsus', ['RSU_0'])[0]
            channel_gain = system_state.get('channel_gains', {}).get(
                (task.vehicle_id, rsu_id), 1e-4)

            trans_time, comp_time = self.calculate_offload_time(
                task, bw_per_task, channel_gain,
                transmit_power, noise_power, comp_per_task
            )
            offload_time = trans_time + comp_time

            # 基于Lyapunov-like决策（但没有相位对齐）
            # 简化版：比较时延
            if offload_time < local_time:
                decision = OffloadingDecision(
                    task_id=task.task_id,
                    offload=True,
                    target_rsu=rsu_id,
                    allocated_bandwidth=bw_per_task,
                    allocated_computation=comp_per_task
                )
            else:
                decision = OffloadingDecision(
                    task_id=task.task_id,
                    offload=False,
                    target_rsu=None,
                    allocated_bandwidth=0.0,
                    allocated_computation=0.0
                )

            decisions.append(decision)

        return decisions

    def get_sync_congestion_level(self, tasks: List[Task],
                                  system_state: Dict) -> float:
        """
        计算同步拥塞程度
        返回：0-1之间的拥塞指数
        """
        if len(tasks) == 0:
            return 0.0

        # 检查有多少任务在同一时刻生成
        generation_times = [task.generation_time for task in tasks]

        # 计算时间方差（方差小表示同步程度高）
        if len(generation_times) > 1:
            variance = np.var(generation_times)
            # 归一化（假设最大方差为周期的1/4）
            max_period = max(task.period for task in tasks)
            normalized_variance = variance / (max_period / 4) ** 2
            congestion = 1 - min(1.0, normalized_variance)
        else:
            congestion = 0.0

        return congestion
```

## 9. 基线算法管理器

```python
"""
baseline_manager.py - 统一管理所有基线算法
"""

from typing import Dict, List, Type
from enum import Enum

class BaselineType(Enum):
    LOCAL_ONLY = "local_only"
    FULL_OFFLOAD = "full_offload"
    GREEDY = "greedy"
    RANDOM = "random"
    POISSON = "poisson"
    NO_PHASE_ALIGN = "no_phase_align"

class BaselineManager:
    """基线算法管理器"""

    # 算法注册表
    ALGORITHM_REGISTRY: Dict[BaselineType, Type[BaselineAlgorithm]] = {
        BaselineType.LOCAL_ONLY: LocalOnlyAlgorithm,
        BaselineType.FULL_OFFLOAD: FullOffloadAlgorithm,
        BaselineType.GREEDY: GreedyOffloadAlgorithm,
        BaselineType.RANDOM: RandomOffloadAlgorithm,
        BaselineType.POISSON: PoissonModelAlgorithm,
        BaselineType.NO_PHASE_ALIGN: NoPhaseAlignAlgorithm,
    }

    def __init__(self, config: Dict):
        self.config = config
        self.algorithms: Dict[BaselineType, BaselineAlgorithm] = {}

    def create_algorithm(self, algo_type: BaselineType) -> BaselineAlgorithm:
        """创建指定类型的基线算法"""
        if algo_type not in self.ALGORITHM_REGISTRY:
            raise ValueError(f"Unknown algorithm type: {algo_type}")

        algo_class = self.ALGORITHM_REGISTRY[algo_type]
        algorithm = algo_class(self.config)
        self.algorithms[algo_type] = algorithm

        return algorithm

    def create_all_algorithms(self) -> Dict[BaselineType, BaselineAlgorithm]:
        """创建所有基线算法"""
        for algo_type in BaselineType:
            self.create_algorithm(algo_type)
        return self.algorithms

    def run_comparison(self, tasks: List[Task],
                      system_state: Dict) -> Dict[str, List[OffloadingDecision]]:
        """运行所有基线算法进行对比"""
        results = {}

        for algo_type, algorithm in self.algorithms.items():
            decisions = algorithm.make_decision(tasks, system_state)
            results[algorithm.name] = decisions

        return results

    def benchmark(self, tasks: List[Task],
                 system_state: Dict,
                 config: Dict) -> Dict[str, Dict]:
        """
        对所有算法进行基准测试

        Returns:
            {算法名: {avg_delay, avg_energy, avg_aoi, success_rate}}
        """
        results = {}

        for algo_type, algorithm in self.algorithms.items():
            decisions = algorithm.make_decision(tasks, system_state)

            # 执行并统计
            total_delay = 0.0
            total_energy = 0.0
            total_aoi = 0.0
            success_count = 0

            for task, decision in zip(tasks, decisions):
                if hasattr(algorithm, 'execute'):
                    if decision.offload:
                        result = algorithm.execute(task, decision, system_state, config)
                    else:
                        result = algorithm.execute(task, config)
                else:
                    # 默认执行逻辑
                    result = self._default_execute(task, decision, system_state, config)

                total_delay += result.completion_time
                total_energy += result.energy_consumption
                total_aoi += result.aoi
                if result.success:
                    success_count += 1

            n_tasks = len(tasks) if len(tasks) > 0 else 1

            results[algorithm.name] = {
                'avg_delay': total_delay / n_tasks,
                'avg_energy': total_energy / n_tasks,
                'avg_aoi': total_aoi / n_tasks,
                'success_rate': success_count / n_tasks
            }

        return results

    def _default_execute(self, task: Task, decision: OffloadingDecision,
                        system_state: Dict, config: Dict) -> ExecutionResult:
        """默认执行逻辑"""
        f_local = config.get('f_local', 1e9)
        kappa = config.get('kappa', 1e-27)
        transmit_power = config.get('transmit_power', 0.5)
        noise_power = config.get('noise_power', 1e-13)

        if not decision.offload:
            # 本地执行
            exec_time = task.computation_load / f_local
            energy = kappa * task.computation_load * (f_local ** 2)
        else:
            # 卸载执行
            channel_gain = system_state.get('channel_gains', {}).get(
                (task.vehicle_id, decision.target_rsu), 1e-4)

            snr = transmit_power * channel_gain / noise_power
            rate = decision.allocated_bandwidth * np.log2(1 + snr)
            trans_time = task.data_size / rate if rate > 0 else float('inf')
            comp_time = task.computation_load / decision.allocated_computation

            exec_time = trans_time + comp_time
            energy = transmit_power * trans_time

        success = exec_time <= task.deadline
        aoi = exec_time if success else task.deadline + exec_time

        return ExecutionResult(
            task_id=task.task_id,
            completion_time=exec_time,
            energy_consumption=energy,
            aoi=aoi,
            success=success
        )
```

```python
"""
baseline_example.py - 基线算法使用示例

def run_baseline_comparison():
    """运行基线算法对比实验"""

    # 配置
    config = {
        'f_local': 1e9,          # 本地计算能力 1 GHz
        'transmit_power': 0.5,    # 传输功率 500 mW
        'noise_power': 1e-13,     # 噪声功率
        'kappa': 1e-27,           # 能耗系数
        'slot_duration': 0.1,     # 时隙长度 100 ms
        'poisson_rate': 10.0,     # 泊松率估计
    }

    # 创建基线管理器
    manager = BaselineManager(config)
    algorithms = manager.create_all_algorithms()

    # 生成测试任务
    tasks = []
    for i in range(10):
        task = Task(
            task_id=f"task_{i}",
            vehicle_id=f"vehicle_{i % 5}",
            data_size=1e6,           # 1 Mbit
            computation_load=1e9,     # 1 Gcycles
            deadline=0.1,             # 100 ms
            generation_time=0.0,
            period=0.1,
            priority=1
        )
        tasks.append(task)

    # 系统状态
    system_state = {
        'current_time': 0.0,
        'total_bandwidth': 20e6,
        'total_computation': 10e9,
        'available_rsus': ['RSU_0', 'RSU_1'],
        'vehicle_positions': {f"vehicle_{i}": (i * 100, 0) for i in range(5)},
        'rsu_positions': {'RSU_0': (0, 50), 'RSU_1': (300, 50)},
        'channel_gains': {
            (f"vehicle_{i}", f"RSU_{j}"): 1e-4 * np.exp(-i * 0.1)
            for i in range(5) for j in range(2)
        }
    }

    # 运行基准测试
    results = manager.benchmark(tasks, system_state, config)

    # 打印结果
    print("\n" + "=" * 60)
    print("基线算法对比结果")
    print("=" * 60)

    headers = ["算法", "平均时延(ms)", "平均能耗(J)", "平均AoI(ms)", "成功率(%)"]
    print(f"{headers[0]:<20} {headers[1]:<15} {headers[2]:<12} {headers[3]:<12} {headers[4]:<10}")
    print("-" * 60)

    for algo_name, metrics in results.items():
        print(f"{algo_name:<20} "
              f"{metrics['avg_delay']*1000:<15.2f} "
              f"{metrics['avg_energy']*1000:<12.4f} "
              f"{metrics['avg_aoi']*1000:<12.2f} "
              f"{metrics['success_rate']*100:<10.1f}")

    return results

if __name__ == "__main__":
    results = run_baseline_comparison()
```
