## 1. 仿真平台选择

### 1.1 方案对比

| 方案 | 优点 | 缺点 | 推荐度 |
|------|------|------|--------|
| **Python自建** | 灵活、快速迭代、易调试 | 需自己实现移动性 | ⭐⭐⭐⭐⭐ |
| SUMO + Python | 真实交通流 | 学习成本高 | ⭐⭐⭐ |
| OMNeT++ / Veins | 完整通信仿真 | 复杂、修改困难 | ⭐⭐ |
| NS-3 | 网络仿真标准 | 学习曲线陡峭 | ⭐⭐ |

**推荐**：使用**Python自建仿真器**，便于快速验证算法。

### 1.2 设计原则

1. **模块化**：各组件独立，便于替换和测试
2. **可配置**：参数通过配置文件管理
3. **可复现**：固定随机种子，结果可重复
4. **可扩展**：易于添加新算法和场景

```python
"""
仿真环境主类
文件：simulation/environment.py
"""

import numpy as np
from typing import List, Dict, Optional
from dataclasses import dataclass

from config.simulation_config import SimulationConfig
from simulation.entities.vehicle import Vehicle
from simulation.entities.rsu import RSU
from simulation.entities.task import Task
from algorithms.phase_allocator import PhaseAllocator
from algorithms.predictor import HybridPredictor
from algorithms.lyapunov_optimizer import LyapunovOptimizer
from algorithms.resource_allocator import ResourceAllocator
from algorithms.migration_manager import MigrationManager
from metrics.metrics import MetricsCollector


class SimulationEnvironment:
    """
    V2X周期性任务卸载仿真环境
    """

    def __init__(self, config: SimulationConfig):
        """
        初始化仿真环境

        Args:
            config: 仿真配置
        """
        self.config = config
        np.random.seed(config.random_seed)

        # 初始化实体
        self.rsus = self._init_rsus()
        self.vehicles = self._init_vehicles()

        # 初始化算法模块
        self.phase_allocator = PhaseAllocator(config.scenario.T_check)
        self.predictor = HybridPredictor(T_check=config.scenario.T_check)
        self.optimizer = LyapunovOptimizer(
            num_vehicles=config.scenario.num_vehicles,
            V=config.algorithm.V,
            A_target=config.algorithm.A_target,
            E_budget_per_slot=config.algorithm.E_budget
        )
        self.resource_allocator = ResourceAllocator()
        self.migration_manager = MigrationManager(
            lookahead_slots=config.algorithm.migration_lookahead,
            margin=config.algorithm.migration_margin
        )

        # 初始化相位
        self.phase_allocator.centralized_allocation(self.vehicles, self.rsus)

        # 统计收集器
        self.metrics = MetricsCollector()

        # 状态变量
        self.current_slot = 0
        self.total_slots = int(config.scenario.total_time / config.scenario.T_check)
        self.warmup_slots = int(config.scenario.warmup_time / config.scenario.T_check)

    def _init_rsus(self) -> List[RSU]:
        """初始化RSU"""
        rsus = []
        cfg = self.config.scenario

        for i in range(cfg.num_rsus):
            x = (i + 0.5) * cfg.rsu_spacing
            rsu = RSU(
                id=i,
                position=np.array([x, cfg.lane_width * cfg.num_lanes / 2]),
                coverage_radius=cfg.rsu_coverage,
                max_freq=self.config.compute.rsu_max_freq,
                bandwidth=self.config.communication.total_bandwidth,
                backhaul_rate=self.config.communication.backhaul_rate
            )
            rsus.append(rsu)

        return rsus

    def _init_vehicles(self) -> List[Vehicle]:
        """初始化车辆"""
        vehicles = []
        cfg = self.config.scenario

        for i in range(cfg.num_vehicles):
            # 随机初始位置
            x = np.random.uniform(0, cfg.road_length)
            lane = np.random.randint(0, cfg.num_lanes)
            y = (lane + 0.5) * cfg.lane_width

            # 随机速度
            speed = np.random.uniform(*cfg.speed_range)

            vehicle = Vehicle(
                id=i,
                position=np.array([x, y]),
                speed=speed,
                heading=0,
                config=self.config
            )
            vehicles.append(vehicle)

        return vehicles

    def step(self) -> Dict:
        """
        执行一个时隙的仿真

        Returns:
            slot_results: 本时隙的结果
        """
        slot = self.current_slot

        # 1. 更新车辆位置
        self._update_vehicle_positions()

        # 2. 任务生成检测
        tasks = {}
        for vehicle in self.vehicles:
            if vehicle.check_task_generation(slot, self.config.task):
                tasks[vehicle.id] = vehicle.current_task

        # 3. 预测
        predictions = {}
        for vehicle in self.vehicles:
            predictions[vehicle.id] = self.predictor.predict(
                vehicle.get_state(),
                K=self.config.algorithm.predict_horizon
            )

        # 4. 信道状态
        channel_states = self._get_channel_states()

        # 5. 卸载决策
        decisions = self.optimizer.make_decisions(
            self.vehicles, self.rsus, tasks, channel_states
        )

        # 6. 资源分配
        for rsu in self.rsus:
            rsu_tasks = [
                tasks[d.vehicle_id]
                for d in decisions
                if d.offload and d.rsu_id == rsu.id and d.vehicle_id in tasks
            ]
            if rsu_tasks:
                allocations = self.resource_allocator.allocate(
                    rsu_tasks, rsu.max_freq, rsu.bandwidth
                )
                rsu.apply_allocation(allocations)

        # 7. 执行任务
        results = self._execute_tasks(tasks, decisions)

        # 8. 迁移检查
        for vehicle in self.vehicles:
            if vehicle.current_rsu is not None:
                decision = self.migration_manager.check_migration(
                    vehicle, vehicle.current_rsu, self.rsus, self.predictor
                )
                if decision is not None:
                    target_rsu = next(r for r in self.rsus if r.id == decision.target_rsu_id)
                    self.migration_manager.execute_migration(
                        decision, vehicle, vehicle.current_rsu, target_rsu
                    )

        # 9. 更新队列
        self.optimizer.update_queues(results, np.array([v.aoi for v in self.vehicles]))

        # 10. 收集指标（跳过预热期）
        if slot >= self.warmup_slots:
            self.metrics.collect(slot, self.vehicles, decisions, results)

        self.current_slot += 1

        return {
            'slot': slot,
            'num_tasks': len(tasks),
            'decisions': decisions,
            'results': results
        }

    def _update_vehicle_positions(self):
        """更新车辆位置"""
        T_check = self.config.scenario.T_check
        road_length = self.config.scenario.road_length

        for vehicle in self.vehicles:
            # 简单直线运动
            vehicle.position[0] += vehicle.speed * T_check

            # 边界处理（循环）
            if vehicle.position[0] > road_length:
                vehicle.position[0] -= road_length

            # 更新可用RSU
            vehicle.update_available_rsus(self.rsus)

    def _get_channel_states(self) -> Dict:
        """获取所有车辆-RSU的信道状态"""
        channel_states = {}
        cfg = self.config.communication

        for vehicle in self.vehicles:
            for rsu in self.rsus:
                distance = np.linalg.norm(vehicle.position - rsu.position)
                if distance <= rsu.coverage_radius:
                    # 计算信道增益和速率
                    path_loss = cfg.h0 * (distance ** (-cfg.path_loss_exp))
                    snr = cfg.tx_power * path_loss / cfg.noise_power
                    rate = cfg.total_bandwidth * np.log2(1 + snr)

                    channel_states[(vehicle.id, rsu.id)] = {
                        'distance': distance,
                        'snr': snr,
                        'rate': rate
                    }

        return channel_states

    def _execute_tasks(self, tasks: Dict, decisions: List) -> List:
        """执行任务并返回结果"""
        from simulation.entities.task import Result

        results = []

        for decision in decisions:
            vehicle = self.vehicles[decision.vehicle_id]

            if decision.vehicle_id not in tasks:
                results.append(Result(
                    vehicle_id=decision.vehicle_id,
                    completed=True,
                    delay=0,
                    energy=0,
                    aoi=vehicle.aoi + self.config.scenario.T_check
                ))
                vehicle.aoi += self.config.scenario.T_check
                continue

            task = tasks[decision.vehicle_id]

            if decision.offload:
                result = self._execute_offload(vehicle, task, decision.rsu_id)
            else:
                result = self._execute_local(vehicle, task)

            # 更新AoI
            if result.completed:
                vehicle.aoi = result.delay
            else:
                vehicle.aoi += self.config.scenario.T_check

            result.aoi = vehicle.aoi
            results.append(result)

            vehicle.clear_task()

        return results

    def _execute_local(self, vehicle, task) -> 'Result':
        """本地执行任务"""
        from simulation.entities.task import Result

        cfg = self.config.compute
        delay = task.compute_load / cfg.local_cpu_freq
        energy = cfg.kappa * (cfg.local_cpu_freq ** 2) * task.compute_load
        completed = delay <= task.deadline

        return Result(
            vehicle_id=vehicle.id,
            completed=completed,
            delay=delay,
            energy=energy,
            aoi=0
        )

    def _execute_offload(self, vehicle, task, rsu_id: int) -> 'Result':
        """卸载执行任务"""
        from simulation.entities.task import Result

        rsu = self.rsus[rsu_id]
        cfg = self.config

        # 计算延迟
        distance = np.linalg.norm(vehicle.position - rsu.position)
        path_loss = cfg.communication.h0 * (distance ** (-cfg.communication.path_loss_exp))
        snr = cfg.communication.tx_power * path_loss / cfg.communication.noise_power
        rate = cfg.communication.total_bandwidth * np.log2(1 + snr) / 10  # 假设10用户共享

        delay_tx = task.data_size / rate
        delay_exe = task.compute_load / (rsu.max_freq / 10)
        delay_down = task.result_size / rate

        delay = delay_tx + delay_exe + delay_down

        # 计算能耗
        energy = (cfg.communication.tx_power * delay_tx +
                  cfg.compute.idle_power * (delay_exe + delay_down))

        completed = delay <= task.deadline

        return Result(
            vehicle_id=vehicle.id,
            completed=completed,
            delay=delay,
            energy=energy,
            aoi=0
        )

    def run(self, num_slots: int = None, verbose: bool = True) -> Dict:
        """
        运行仿真

        Args:
            num_slots: 运行时隙数
            verbose: 是否打印进度

        Returns:
            final_metrics: 最终统计指标
        """
        if num_slots is None:
            num_slots = self.total_slots

        for _ in range(num_slots):
            self.step()

            if verbose and self.current_slot % self.config.log_interval == 0:
                print(f"Slot {self.current_slot}/{num_slots}")

        return self.metrics.summarize()

    def reset(self):
        """重置仿真环境"""
        np.random.seed(self.config.random_seed)

        self.vehicles = self._init_vehicles()
        self.phase_allocator.centralized_allocation(self.vehicles, self.rsus)
        self.optimizer.reset()
        self.metrics.reset()
        self.current_slot = 0
```

### 4.1 指标定义

```python
"""
性能指标收集与计算
文件：metrics/metrics.py
"""

import numpy as np
from typing import List, Dict
from collections import defaultdict


class MetricsCollector:
    """
    性能指标收集器
    """

    def __init__(self):
        self.reset()

    def reset(self):
        """重置所有指标"""
        self.aoi_history = []           # AoI历史
        self.delay_history = []         # 延迟历史
        self.energy_history = []        # 能耗历史
        self.completion_history = []    # 完成率历史
        self.offload_ratio_history = [] # 卸载率历史
        self.migration_count = 0        # 迁移次数
        self.task_count = 0             # 任务总数

    def collect(self, slot: int, vehicles: List, decisions: List, results: List):
        """
        收集单个时隙的指标

        Args:
            slot: 当前时隙
            vehicles: 车辆列表
            decisions: 决策列表
            results: 执行结果列表
        """
        # AoI
        aois = [v.aoi for v in vehicles]
        self.aoi_history.append(np.mean(aois))

        # 延迟、能耗、完成率
        delays = []
        energies = []
        completions = []

        for result in results:
            if result.delay > 0:
                delays.append(result.delay)
                energies.append(result.energy)
                completions.append(1 if result.completed else 0)
                self.task_count += 1

        if delays:
            self.delay_history.append(np.mean(delays))
            self.energy_history.append(np.mean(energies))
            self.completion_history.append(np.mean(completions))

        # 卸载率
        offload_count = sum(1 for d in decisions if d.offload)
        total_count = len([d for d in decisions if d.expected_delay > 0])
        if total_count > 0:
            self.offload_ratio_history.append(offload_count / total_count)

    def summarize(self) -> Dict:
        """
        汇总所有指标

        Returns:
            summary: 指标汇总字典
        """
        summary = {
            # 平均值
            'avg_aoi': np.mean(self.aoi_history) if self.aoi_history else 0,
            'avg_delay': np.mean(self.delay_history) if self.delay_history else 0,
            'avg_energy': np.mean(self.energy_history) if self.energy_history else 0,
            'avg_completion_rate': np.mean(self.completion_history) if self.completion_history else 0,
            'avg_offload_ratio': np.mean(self.offload_ratio_history) if self.offload_ratio_history else 0,

            # 标准差
            'std_aoi': np.std(self.aoi_history) if self.aoi_history else 0,
            'std_delay': np.std(self.delay_history) if self.delay_history else 0,

            # 百分位数
            'p95_aoi': np.percentile(self.aoi_history, 95) if self.aoi_history else 0,
            'p95_delay': np.percentile(self.delay_history, 95) if self.delay_history else 0,

            # 计数
            'total_tasks': self.task_count,
            'migration_count': self.migration_count,

            # 原始数据（用于绘图）
            'aoi_history': self.aoi_history,
            'delay_history': self.delay_history,
            'completion_history': self.completion_history
        }

        return summary

    def print_summary(self):
        """打印指标摘要"""
        summary = self.summarize()

        print("\n" + "=" * 50)
        print("性能指标汇总")
        print("=" * 50)
        print(f"平均AoI:        {summary['avg_aoi']*1000:.2f} ms")
        print(f"平均延迟:       {summary['avg_delay']*1000:.2f} ms")
        print(f"平均能耗:       {summary['avg_energy']*1000:.3f} mJ")
        print(f"任务完成率:     {summary['avg_completion_rate']*100:.1f}%")
        print(f"卸载率:         {summary['avg_offload_ratio']*100:.1f}%")
        print(f"95%延迟:        {summary['p95_delay']*1000:.2f} ms")
        print(f"总任务数:       {summary['total_tasks']}")
        print("=" * 50)
```

```python
文件：experiments/exp1_baseline_comparison.py
"""

import numpy as np
import matplotlib.pyplot as plt
from config.simulation_config import SimulationConfig
from simulation.environment import SimulationEnvironment

# 基线算法
from baselines.local_only import LocalOnlyBaseline
from baselines.full_offload import FullOffloadBaseline
from baselines.greedy import GreedyBaseline
from baselines.random_offload import RandomOffloadBaseline


def run_baseline_comparison():
    """运行基线对比实验"""

    config = SimulationConfig()
    config.scenario.total_time = 500.0  # 500秒

    algorithms = {
        'ETPO (Proposed)': None,  # 使用默认算法
        'Local Only': LocalOnlyBaseline(),
        'Full Offload': FullOffloadBaseline(),
        'Greedy': GreedyBaseline(),
        'Random': RandomOffloadBaseline()
    }

    results = {}

    for name, algorithm in algorithms.items():
        print(f"\n运行 {name}...")

        env = SimulationEnvironment(config)

        if algorithm is not None:
            env.optimizer = algorithm  # 替换优化器

        metrics = env.run(verbose=False)
        results[name] = metrics

        print(f"  平均AoI: {metrics['avg_aoi']*1000:.2f} ms")
        print(f"  完成率: {metrics['avg_completion_rate']*100:.1f}%")

    # 绘图
    plot_comparison(results)

    return results


def plot_comparison(results: Dict):
    """绘制对比图"""

    algorithms = list(results.keys())
    metrics_to_plot = ['avg_aoi', 'avg_delay', 'avg_completion_rate', 'avg_energy']
    titles = ['平均AoI (ms)', '平均延迟 (ms)', '任务完成率 (%)', '平均能耗 (mJ)']
    multipliers = [1000, 1000, 100, 1000]

    fig, axes = plt.subplots(2, 2, figsize=(12, 10))
    axes = axes.flatten()

    for idx, (metric, title, mult) in enumerate(zip(metrics_to_plot, titles, multipliers)):
        values = [results[alg][metric] * mult for alg in algorithms]

        bars = axes[idx].bar(range(len(algorithms)), values, color='steelblue')
        bars[0].set_color('coral')  # 突出显示提出的方法

        axes[idx].set_xticks(range(len(algorithms)))
        axes[idx].set_xticklabels(algorithms, rotation=45, ha='right')
        axes[idx].set_ylabel(title)
        axes[idx].set_title(title)

        # 添加数值标签
        for bar, val in zip(bars, values):
            axes[idx].text(bar.get_x() + bar.get_width()/2, bar.get_height(),
                          f'{val:.1f}', ha='center', va='bottom', fontsize=9)

    plt.tight_layout()
    plt.savefig('results/exp1_baseline_comparison.png', dpi=150)
    plt.show()


if __name__ == "__main__":
    run_baseline_comparison()
```

```python
"""
实验2：参数敏感性分析
文件：experiments/exp2_parameter_sensitivity.py
"""

def run_parameter_sensitivity():
    """参数敏感性分析"""

    config = SimulationConfig()
    config.scenario.total_time = 200.0

    # 参数范围
    V_values = [10, 50, 100, 200, 500, 1000]
    vehicle_counts = [20, 40, 60, 80, 100]
    speed_ranges = [(5, 15), (10, 25), (15, 35), (20, 40)]

    # 实验1：V参数影响
    print("\n=== V参数敏感性 ===")
    v_results = {'V': [], 'aoi': [], 'completion': []}

    for V in V_values:
        config.algorithm.V = V
        env = SimulationEnvironment(config)
        metrics = env.run(verbose=False)

        v_results['V'].append(V)
        v_results['aoi'].append(metrics['avg_aoi'] * 1000)
        v_results['completion'].append(metrics['avg_completion_rate'] * 100)

        print(f"V={V}: AoI={metrics['avg_aoi']*1000:.2f}ms, "
              f"完成率={metrics['avg_completion_rate']*100:.1f}%")

    # 实验2：车辆数量影响
    print("\n=== 车辆数量敏感性 ===")
    vehicle_results = {'count': [], 'aoi': [], 'completion': []}

    config.algorithm.V = 100  # 恢复默认

    for count in vehicle_counts:
        config.scenario.num_vehicles = count
        env = SimulationEnvironment(config)
        metrics = env.run(verbose=False)

        vehicle_results['count'].append(count)
        vehicle_results['aoi'].append(metrics['avg_aoi'] * 1000)
        vehicle_results['completion'].append(metrics['avg_completion_rate'] * 100)

        print(f"车辆数={count}: AoI={metrics['avg_aoi']*1000:.2f}ms, "
              f"完成率={metrics['avg_completion_rate']*100:.1f}%")

    return v_results, vehicle_results


if __name__ == "__main__":
    run_parameter_sensitivity()
```
